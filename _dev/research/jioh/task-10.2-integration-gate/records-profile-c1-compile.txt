## The script

import cProfile, pstats, sys, time, pathlib, resource
sys.path.insert(0, "harness/tools")
from harness import records, scoring
from harness import aggregates as agg
REPO = pathlib.Path(".")   # the repository root
run = REPO / "runs/integration-gate/smoke/c1-compile/fixed~focus"
wl = REPO / "dataset/build/coreset-single/c1-compile.workload.json"
spec = scoring.load_spec(REPO / "harness/scoring/scoring-spec.yaml")
prof = cProfile.Profile()
t0 = time.time()
prof.enable()
rows, msgs = records.build(wl, run / "trace.jsonl", schedule_path=run / "schedule.json", telemetry="focus")
prof.disable()
t1 = time.time()
records.write_csv(rows, pathlib.Path(sys.argv[1]) / "records.csv")
t2 = time.time()
a = agg.compute_aggregates(rows, scoring.windows_from_spec(spec, "c1-compile"), scoring.interactive_from_spec(spec, "c1-compile"))
t3 = time.time()
print(f"rows {len(rows)}; records.build {t1-t0:.0f} s; write_csv {t2-t1:.0f} s; aggregates {t3-t2:.0f} s; peak RSS {resource.getrusage(resource.RUSAGE_SELF).ru_maxrss/1e9:.2f} GB")
pstats.Stats(prof).sort_stats("cumulative").print_stats(25)
